# create a service registration file 
sudo nano /etc/systemd/system/violation-detection-system.service

# add the below content
[Unit]
Description=Drive Safe Application
After=network.target

[Service]
ExecStart=/usr/bin/python3 /home/pi/Desktop/violation-detection-system/main.py
WorkingDirectory=/home/pi/Desktop/violation-detection-system
User=pi
Group=pi
Restart=always
RestartSec=5
Environment=DISPLAY=:0
Environment=PYTHONUNBUFFERED=1

[Install]
WantedBy=multi-user.target


# execute this 
sudo systemctl daemon-reload

# execute this to enable or activate the service to be loaded at boot time  
sudo systemctl enable /home/pi/Desktop/violation-detection-system.service

# start the service without rebooting the system 
sudo systemctl start violation-detection-system.service

# stop the service without rebooting the system 
sudo systemctl stop violation-detection-system.service

# check the service status 
sudo systemctl status violation-detection-system.service
